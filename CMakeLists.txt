cmake_minimum_required(VERSION 3.28)
project(Gatehaven VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

option(GATEHAVEN_BUILD_TESTS "Build core tests" ON)
option(GATEHAVEN_SANITIZERS "Enable address and undefined-behavior sanitizers" OFF)
option(GATEHAVEN_WARNINGS_AS_ERRORS "Treat project warnings as errors" ON)

function(gatehaven_defaults target)
    target_compile_features(${target} PUBLIC cxx_std_23)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
        if(GATEHAVEN_WARNINGS_AS_ERRORS)
            target_compile_options(${target} PRIVATE /WX)
        endif()
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
        if(GATEHAVEN_WARNINGS_AS_ERRORS)
            target_compile_options(${target} PRIVATE -Werror)
        endif()
        if(GATEHAVEN_SANITIZERS)
            target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(${target} PRIVATE -fsanitize=address,undefined)
        endif()
    endif()
endfunction()

add_library(gatehaven_core src/core/circuit.cpp src/core/simulation.cpp src/core/document.cpp src/core/editor.cpp)
target_include_directories(gatehaven_core PUBLIC include)
gatehaven_defaults(gatehaven_core)

if(GATEHAVEN_BUILD_TESTS)
    enable_testing()
    add_executable(gatehaven_tests tests/main.cpp tests/circuit_tests.cpp tests/simulation_tests.cpp tests/document_tests.cpp tests/editor_tests.cpp)
    target_link_libraries(gatehaven_tests PRIVATE gatehaven_core)
    gatehaven_defaults(gatehaven_tests)
    add_test(NAME gatehaven_core COMMAND gatehaven_tests)
endif()

cmake_minimum_required(VERSION 3.28)
project(Gatehaven VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

option(GATEHAVEN_BUILD_TESTS "Build core tests" ON)
option(GATEHAVEN_BUILD_APP "Build the SDL3 desktop application" OFF)
option(GATEHAVEN_BUILD_BENCHMARKS "Build repeatable simulation workloads" OFF)
option(GATEHAVEN_SANITIZERS "Enable address and undefined-behavior sanitizers" OFF)
option(GATEHAVEN_WARNINGS_AS_ERRORS "Treat project warnings as errors" ON)

function(gatehaven_defaults target)
    target_compile_features(${target} PUBLIC cxx_std_23)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
        if(GATEHAVEN_WARNINGS_AS_ERRORS)
            target_compile_options(${target} PRIVATE /WX)
        endif()
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
        if(GATEHAVEN_WARNINGS_AS_ERRORS)
            target_compile_options(${target} PRIVATE -Werror)
        endif()
        if(GATEHAVEN_SANITIZERS)
            target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(${target} PRIVATE -fsanitize=address,undefined)
        endif()
    endif()
endfunction()

add_library(gatehaven_core src/core/circuit.cpp src/core/simulation.cpp src/core/document.cpp src/core/editor.cpp src/core/examples.cpp src/core/clipboard_codec.cpp)
target_sources(gatehaven_core PRIVATE src/core/polyline.cpp)
target_sources(gatehaven_core PRIVATE src/core/selection.cpp)
target_include_directories(gatehaven_core PUBLIC include)
gatehaven_defaults(gatehaven_core)

find_package(Threads REQUIRED)
add_library(gatehaven_platform src/platform/clipboard_session.cpp src/platform/process.cpp)
target_link_libraries(gatehaven_platform PUBLIC gatehaven_core PRIVATE Threads::Threads)
gatehaven_defaults(gatehaven_platform)

add_executable(gatehaven-cli src/cli.cpp)
target_link_libraries(gatehaven-cli PRIVATE gatehaven_core)
gatehaven_defaults(gatehaven-cli)

if(GATEHAVEN_BUILD_BENCHMARKS)
    add_executable(gatehaven-bench bench/simulation_bench.cpp)
    target_link_libraries(gatehaven-bench PRIVATE gatehaven_core)
    gatehaven_defaults(gatehaven-bench)
endif()

if(GATEHAVEN_BUILD_APP)
    find_package(SDL3 3.4.18 EXACT CONFIG REQUIRED)
    add_executable(gatehaven src/app/main.cpp src/app/font.cpp src/app/instances.cpp)
    target_link_libraries(gatehaven PRIVATE gatehaven_platform SDL3::SDL3)
    gatehaven_defaults(gatehaven)
    add_custom_target(gatehaven_snapshot
        COMMAND gatehaven --snapshot ${CMAKE_CURRENT_BINARY_DIR}/gatehaven.bmp
        DEPENDS gatehaven VERBATIM)
    if(APPLE)
        set_target_properties(gatehaven PROPERTIES MACOSX_BUNDLE TRUE
            MACOSX_BUNDLE_GUI_IDENTIFIER "com.michaelbaffourawuah.gatehaven"
            MACOSX_BUNDLE_BUNDLE_NAME "Gatehaven")
    endif()
    if(WIN32)
        add_custom_command(TARGET gatehaven POST_BUILD
            COMMAND ${CMAKE_COMMAND} "-DGH_RUNTIME_DLLS=$<JOIN:$<TARGET_RUNTIME_DLLS:gatehaven>,|>"
                "-DGH_DEST=$<TARGET_FILE_DIR:gatehaven>" -P ${CMAKE_CURRENT_SOURCE_DIR}/cmake/copy_runtime.cmake
            VERBATIM)
    endif()
endif()

if(GATEHAVEN_BUILD_TESTS)
    enable_testing()
    add_executable(gatehaven_tests tests/main.cpp tests/circuit_tests.cpp tests/simulation_tests.cpp tests/document_tests.cpp tests/editor_tests.cpp tests/example_tests.cpp tests/viewport_tests.cpp tests/clipboard_codec_tests.cpp tests/process_tests.cpp)
    target_link_libraries(gatehaven_tests PRIVATE gatehaven_platform)
    target_sources(gatehaven_tests PRIVATE tests/polyline_tests.cpp)
    target_sources(gatehaven_tests PRIVATE tests/selection_tests.cpp)
    gatehaven_defaults(gatehaven_tests)
    add_test(NAME gatehaven_core COMMAND gatehaven_tests)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_executable(gatehaven_clipboard_peer tests/clipboard_peer.cpp)
    target_link_libraries(gatehaven_clipboard_peer PRIVATE gatehaven_platform)
    gatehaven_defaults(gatehaven_clipboard_peer)
    add_test(NAME gatehaven_shared_clipboard COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/tests/clipboard_process_test.py $<TARGET_FILE:gatehaven_clipboard_peer>)
    set_tests_properties(gatehaven_shared_clipboard PROPERTIES TIMEOUT 60)
    if(GATEHAVEN_BUILD_APP)
        add_test(NAME gatehaven_desktop COMMAND gatehaven --self-test)
        set_tests_properties(gatehaven_desktop PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy" TIMEOUT 45)
    endif()
endif()
